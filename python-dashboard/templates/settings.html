Python (requests)
import requests

res = requests.post(
    "{{ config.base_url }}/send-sms",
    json={"to": "+1234567890", "message": "Verification code: 849201", "sim_slot": 0}{% if config.auth_enabled %},
    headers={"Authorization": "Bearer {{ config.api_key }}"}{% endif %}
)
print(res.json())

  },

}).then(r => r.json()).then(console.log);
